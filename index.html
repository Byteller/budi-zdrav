<script>

  const fields = {

    name: document.getElementById('in-name'),

    title: document.getElementById('in-title'),

    services: document.getElementById('in-services'),

    phone: document.getElementById('in-phone'),

    email: document.getElementById('in-email'),

    location: document.getElementById('in-location'),

    instagram: document.getElementById('in-instagram'),

    logo: document.getElementById('in-logo')

  };


  function updateSignature() {

    // Ime

    document.getElementById('out-name').textContent = fields.name.value.trim();


    // Titula

    const titleEl = document.getElementById('out-title');

    titleEl.textContent = fields.title.value.trim();

    titleEl.style.display = fields.title.value.trim() ? 'block' : 'none';


    // Djelatnost

    const servEl = document.getElementById('out-services');

    servEl.textContent = fields.services.value.trim();

    servEl.style.display = fields.services.value.trim() ? 'block' : 'none';


    // Telefon

    const phoneVal = fields.phone.value.trim();

    const phoneEl = document.getElementById('out-phone');

    const rowPhone = document.getElementById('row-phone');

    if (phoneVal) {

      phoneEl.textContent = phoneVal;

      phoneEl.href = 'tel:' + phoneVal.replace(/[^+\d]/g, '');

      rowPhone.style.display = 'block';

    } else {

      rowPhone.style.display = 'none';

    }


    // Email

    const emailVal = fields.email.value.trim();

    const emailEl = document.getElementById('out-email');

    const rowEmail = document.getElementById('row-email');

    if (emailVal) {

      emailEl.textContent = emailVal;

      emailEl.href = 'mailto:' + emailVal;

      rowEmail.style.display = 'block';

    } else {

      rowEmail.style.display = 'none';

    }


    // Lokacija

    const locVal = fields.location.value.trim();

    const locEl = document.getElementById('out-location');

    locEl.textContent = locVal;

    locEl.style.display = locVal ? 'block' : 'none';


    // Instagram

    const igVal = fields.instagram.value.trim();

    const igEl = document.getElementById('out-instagram');

    const rowIg = document.getElementById('row-instagram');

    if (igVal) {

      igEl.href = igVal;

      rowIg.style.display = 'block';

    } else {

      rowIg.style.display = 'none';

    }


    // Logo

    const logoVal = fields.logo.value.trim();

    const logoImg = document.getElementById('logo-img-fixed');

    if (logoVal) {

      logoImg.src = logoVal;

      logoImg.style.display = 'block';

    } else {

      logoImg.style.display = 'none';

    }

  }


  Object.values(fields).forEach(input => {

    input.addEventListener('input', updateSignature);

  });


  // Upload i crop profilne slike u kvadrat (76x76)

  document.getElementById('in-photo').addEventListener('change', function(e) {

    const file = e.target.files[0];

    if (!file) return;


    const reader = new FileReader();

    reader.onload = function(event) {

      const img = new Image();

      img.onload = function() {

        const canvas = document.createElement('canvas');

        const size = 152; // 2x za retina ekrane

        canvas.width = size;

        canvas.height = size;

        const ctx = canvas.getContext('2d');


        const minSide = Math.min(img.width, img.height);

        const sx = (img.width - minSide) / 2;

        const sy = (img.height - minSide) / 2;


        ctx.drawImage(img, sx, sy, minSide, minSide, 0, 0, size, size);


        const dataUrl = canvas.toDataURL('image/jpeg', 0.9);

        document.getElementById('photo-img').src = dataUrl;

        document.getElementById('cell-photo').style.display = 'table-cell';

        document.getElementById('btn-remove-photo').style.display = 'inline-block';

      };

      img.src = event.target.result;

    };

    reader.readAsDataURL(file);

  });


  function removePhoto() {

    document.getElementById('in-photo').value = '';

    document.getElementById('photo-img').src = '';

    document.getElementById('cell-photo').style.display = 'none';

    document.getElementById('btn-remove-photo').style.display = 'none';

  }


  // Kopiranje vizuelnog potpisa (za direktan paste u Gmail/Outlook)

  function copySignature() {

    const sigTable = document.getElementById('sig-table');

    const range = document.createRange();

    range.selectNode(sigTable);

    const selection = window.getSelection();

    selection.removeAllRanges();

    selection.addRange(range);

    document.execCommand('copy');

    selection.removeAllRanges();


    const btn = document.getElementById('btn-copy-visual');

    const orig = btn.textContent;

    btn.textContent = 'Kopirano! Zalijepi u Gmail (Ctrl+V)';

    setTimeout(() => btn.textContent = orig, 2500);

  }


  // Kopiranje čistog HTML koda

  function copyHTML() {

    const html = document.getElementById('signature-wrapper').innerHTML.trim();

    navigator.clipboard.writeText(html).then(() => {

      const btn = document.getElementById('btn-copy-html');

      const orig = btn.textContent;

      btn.textContent = 'HTML kopiran!';

      setTimeout(() => btn.textContent = orig, 2000);

    });

  }


  updateSignature();
